   ```sql
   FROM quota_remaining qr
   FULL OUTER JOIN efish_account_balance ef
       ON qr.org_id = ef.org_id
       AND qr.llp = ef.llp
       AND qr.species_code = ef.species_code
       AND qr.year = ef.year
   ```
